import Gtk from 'gi://Gtk';
import GObject from 'gi://GObject';

// List every property on Gtk.Popover (includes inherited from Gtk.Widget)
// NB: Props work the same as html elements attributes. 

for (const pspec of Gtk.Popover.list_properties()) {
    console.log(`Found property: ${pspec.name}`);
}

// Look up a specific property by canonical name
const hasArrow = Gtk.Popover.find_property('has-arrow');
if (hasArrow instanceof GObject.ParamSpec) {
    console.log(`Found property: ${hasArrow.name}`);
}
